<script>
function canedit( o, en=1 ) {
    if ( o instanceof Object ) {
        if ( Array.isArray( o ) ) {
            return canedit.these( o, en );
        }
        if ( o instanceof HTMLElement ) {
            if ( en ) {
                o.setAttribute( _CE, "" );
            } else {
                o.removeAttribute( _CE, "" );
            }
            return;
        }
        return;
    }
    return canedit.all( o, en )
}
;
; _CE = "contenteditable"
;
</script>

<script>
canedit.these = function( list, en=1 ) {
    list.forEach(
        ( ge ) => ( canedit( ge, en ) )
    );
};
</script>

<script>
canedit.all = function( type, en=1 ) {
    const m = all( type );
    canedit . these( m, en );
};
</script>

<script>
function schema( table ) {
    function again( o ) {
        return schema.again( o );
    }
    const type = table.nodeName;
    switch ( type ) {
    case "TABLE" : return schema( table.tHead );
    case "THEAD" : return schema( table.rows[ 0 ] );
    case "TH"    :
    case "TD"    : return schema( table.parentElement );
    }
    if ( type !== "TR" ) {
        throw new Error( `Can't locate table row` );
    }
    return read_row( table );
}
schema.again = function( table ) {
    return this( table );
};
</script>

<script>
function read_row( tr ) {
    const type = tr.nodeName;
    if ( type !== "TR" ) {
        throw new Error( `Can't locate table row` );
    }
    return (
        arr( tr.cells )
        . map( ( ce ) => ( ce.textContent ) )
    );
}
</script>

<script>
function zap( sender ) {
    if ( sender instanceof Event ) {
        sender = ( sender.target );
    }
    const table  = zap.table( sender );
    const action = sender.innerText.trim();
    switch( action.toUpperCase() ) {
    case "➕"  : return zap.add   ( table );
    case "➖"  : return zap.remove( table );
    case "💾" : return zap.save  ( table );
    }
    throw new Error( `Unknown Action : "${action}"` );
}
</script>

<script>
zap.table = function( sender ) {
    function index( ge ) {
        return parseInt( ge.getAttribute( "table" ) );
    }
    let i = index( sender );
    if (! isFinite( i ) ) {
        const pe = sender.parentElement;
        i = index( pe );
        if (! isFinite( i ) ) {
            console.warn( "Invalid Sender", sender );
            throw new Error( `Unable to locate Table` );
        }
    }
    const m = all( "TABLE" );
    return ( m[ i ] );
}
</script>

<script>
zap.add = function( table ) {
    try {
        const k = input_name( "." );
        if (! k ) { return; }
        const be = table.tBodies[ 0 ];
        const re = be . insertRow();
        const sch    = schema( table );
        const fields = sch.length;
        let i, ce;
        for ( i = 0; i < fields; i += 1 ) {
            ce = re.insertCell();
            if ( i > 0 ) {
                ce . textContent = ".";
            } else {
                ce . textContent = ( k );
            }
            canedit( ce );
        }
    } catch ( e ) {
        alert ( e );
        console . error( e );
    }
};
</script>

<script>
zap.remove = function( table ) {
    try {
        const k = input_name( "." );
        if (! k ) { return; }
        const be = table.tBodies[ 0 ];
        let t, ce, rows = arr( be.rows );
        rows = ( rows ).filter(
            ( row ) => {
                ce = row.cells[ 0 ];
                t = str( ce.textContent );
                return str( k === t );
            }
        );
        let re, n = 0;
        while ( rows.length ) {
            re = rows.pop();
            re . remove();
            n += 1;
        }
        alert( `Row(s) Deleted : ${n}` );
    } catch ( e ) {
        alert ( e );
        console . error( e );
    }
};
</script>

<script>
zap.save = function( table ) {
    try {
        const k = input_name(
            "snek-global-table.html", "File Name?"
        );
        if (! k ) { return; }
        const v = table.outerHTML;
        rico( k, v );
    } catch ( e ) {
        alert ( e );
        console . error( e );
    }
};
</script>

<script>
function rico( k, v ) {
    const o = rico.options;
    const b = new Blob( [ v ], o );

    a . download = ( k );

}

};
</script>
