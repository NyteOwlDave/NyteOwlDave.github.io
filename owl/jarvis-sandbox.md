gideon . choose = function( type, index, owner ) {
    function range_error() {
        throw new Error( `Index is Out of Bounds` );
    }
    function type_error() {
        throw new Error( `Invalid Index` );
    }
    const t = str( type );
    if (! t ) {
        return type_error();
    }
    const o = ( owner || doc.body );
    const m = ale( type, o );
    const i = parseInt( index );
    if (! isFinite( i ) ) {
        return range_error();
    }
    if ( ( i >= m.length ) || ( i < 0 ) ) {
        return range_error();
    }
    return ( m[ i ] );
};
</script>

<script>
gideon . table = function( title, schema, records ) {
    function cell( re, text="?", type="TD" ) {
        const ce = elx( type );
        ce . textContent = ( str( text ) || "?" );
        ce . classList . add( "left-align" );
        canedit( ce );
        re . appendChild( ce );
    }
    function row( owner, items=[], type="TD" ) {
        if (! Array.isArray( items ) ) {
            return;
        }
        const re = owner . insertRow();
        items . forEach(
            ( text ) => {
                cell( re, text, type );
            }
        );
    }
    const te = elx( "TABLE" );
    const he = te . createTHead();
    const be = te . createTBody();
    schema  = arr( schema  );
    records = arr( records );
    row( he, schema, "TH" );
    records. forEach(
        ( rec ) => {
            row( be, rec );
        }
    );
    return gideon.fieldset( title, te );
};
</script>

<script>
function fieldset_from_table( table, title ) {
    function missing() {
        throw new Error( `Missing table argument` );
    }
    function bad() {
        throw new Error( `Expected a TABLE Gadget` );
    }
    const te = gideon( table );
    if (! te ) { missing(); }
    if ( te . nodeName !== "TABLE" ) { bad(); }
    const dad = ( te . parentElement );
    if ( dad . classList . contains( "content" ) ) {
        return ( dad . parentElement );
    }
    let t = str( title );
    const sib = ( te . previousElementSibling );
    if ( sib ) {
        const h = ( gideon . headers );
        if ( h . includes( sib . nodeName ) ) {
            t = ( ( t ) || str( sib . textContent ) );
        }
    }
    t = ( t || "Untitled Table" );
    const fe = gideon . fieldset( t, te );
    const owner = (
         gideon( "table_section" )
      || gideon( "table-section" )
      || ( doc . body )
    );
    if ( sib ) { sib . remove(); }
    return owner.appendChild( fe );
};
</script>
